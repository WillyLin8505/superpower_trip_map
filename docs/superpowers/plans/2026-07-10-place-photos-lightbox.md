# Place Photos Lightbox Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add up to four Google photos per place and an in-app lightbox for itinerary and recommendation cards.

**Architecture:** Extend the `Place` model with optional `photoUrls`, normalize Google photo references at the Places action boundary, and render photos through focused `PhotoStrip` and `PhotoLightbox` components. Keep existing `photoUrl` behavior as a fallback so existing fixtures and saved trips continue to work.

**Tech Stack:** Next.js App Router, React client components, TypeScript, Jest, React Testing Library, Google Places Photo proxy at `/api/photo`.

## Global Constraints

- Do not expose `GOOGLE_MAPS_API_KEY` to the browser.
- Keep `photoUrl: string | null` backward compatible.
- Add only optional fields to shared types.
- Do not modify drag, lock, time, day-count, or recommendation-center behavior.
- Use TDD: every production change starts with a failing test.
- No new runtime dependency.

Create `components/PhotoLightbox.tsx`:

```tsx
'use client'
import { useEffect, useState } from 'react'

interface Props {
  photos: string[]
  placeName: string
  initialIndex: number
  onClose: () => void
}

export function PhotoLightbox({ photos, placeName, initialIndex, onClose }: Props) {
  const [index, setIndex] = useState(initialIndex)
  const hasMany = photos.length > 1

  useEffect(() => {
    function onKeyDown(event: KeyboardEvent) {
      if (event.key === 'Escape') onClose()
      if (event.key === 'ArrowRight' && hasMany) setIndex((value) => (value + 1) % photos.length)
      if (event.key === 'ArrowLeft' && hasMany) setIndex((value) => (value - 1 + photos.length) % photos.length)
    }
    document.addEventListener('keydown', onKeyDown)
    return () => document.removeEventListener('keydown', onKeyDown)
  }, [hasMany, onClose, photos.length])

  if (photos.length === 0) return null

  return (
    <div
      role="dialog"
      aria-modal="true"
      aria-label={`${placeName}照片`}
      className="fixed inset-0 z-50 bg-black/70 flex items-center justify-center p-4"
      onClick={onClose}
    >
      <div className="relative max-w-3xl w-full" onClick={(event) => event.stopPropagation()}>
        <button
          type="button"
          aria-label="關閉照片"
          onClick={onClose}
          className="absolute -top-10 right-0 text-white text-sm px-3 py-1 rounded-full bg-black/40 hover:bg-black/60"
        >
          關閉
        </button>
        <img
          src={photos[index]}
          alt={`${placeName} large photo ${index + 1}`}
          className="max-h-[80vh] w-full object-contain rounded-2xl bg-white"
        />
        {hasMany && (
          <div className="absolute inset-y-0 left-0 right-0 flex items-center justify-between px-2">
            <button type="button" aria-label="上一張照片" onClick={() => setIndex((value) => (value - 1 + photos.length) % photos.length)} className="rounded-full bg-white/85 px-3 py-2 text-lg">‹</button>
            <button type="button" aria-label="下一張照片" onClick={() => setIndex((value) => (value + 1) % photos.length)} className="rounded-full bg-white/85 px-3 py-2 text-lg">›</button>
          </div>
        )}
      </div>
    </div>
  )
}
```

- [ ] **Step 4: Implement PhotoStrip**

Create `components/PhotoStrip.tsx`:

```tsx
'use client'
import { useState } from 'react'
import { PhotoLightbox } from './PhotoLightbox'

interface Props {
  photos: string[]
  placeName: string
  className?: string
}

export function PhotoStrip({ photos, placeName, className = '' }: Props) {
  const visiblePhotos = photos.slice(0, 4)
  const [selectedIndex, setSelectedIndex] = useState<number | null>(null)

  if (visiblePhotos.length === 0) return null

  return (
    <>
      <div className={`grid grid-cols-4 gap-1 mt-2 ${className}`}>
        {visiblePhotos.map((photo, index) => (
          <button
            key={`${photo}-${index}`}
            type="button"
            data-testid={`photo-thumb-${index}`}
            aria-label={`查看${placeName}照片 ${index + 1}`}
            onClick={() => setSelectedIndex(index)}
            className="overflow-hidden rounded-lg bg-stone-100 aspect-square border border-white/70 hover:opacity-90 focus:outline-none focus:ring-2 focus:ring-clay"
          >
            <img src={photo} alt={`${placeName} photo ${index + 1}`} className="h-full w-full object-cover" />
          </button>
        ))}
      </div>
      {selectedIndex !== null && (
        <PhotoLightbox
          photos={visiblePhotos}
          placeName={placeName}
          initialIndex={selectedIndex}
          onClose={() => setSelectedIndex(null)}
        />
      )}
    </>
  )
}
```
